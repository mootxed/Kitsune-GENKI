  if (headerEl) {
    headerEl.style.display = header ? '' : 'none';

  function bindEvents() {

        );
        const preview = previewStudyPlanFromPreferences(draft, catalog, { state });

          }

            boundary >= 0 ? courseLessons.slice(0, boundary + 1).map((item) => item.id) : [];
          draft.priorKnowledgeChapterIds = priors;
